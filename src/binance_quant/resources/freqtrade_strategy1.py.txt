from __future__ import annotations

import pandas as pd
from pandas import DataFrame

from freqtrade.strategy import IStrategy, informative


def _ema(series: pd.Series, period: int) -> pd.Series:
    return series.ewm(span=period, adjust=False, min_periods=period).mean()


def _atr(dataframe: DataFrame, period: int = 14) -> pd.Series:
    previous_close = dataframe["close"].shift(1)
    true_range = pd.concat(
        (
            dataframe["high"] - dataframe["low"],
            (dataframe["high"] - previous_close).abs(),
            (dataframe["low"] - previous_close).abs(),
        ),
        axis=1,
    ).max(axis=1)
    return true_range.ewm(alpha=1 / period, adjust=False, min_periods=period).mean()


def _adx(dataframe: DataFrame, period: int = 14) -> pd.Series:
    upward = dataframe["high"].diff()
    downward = -dataframe["low"].diff()
    plus_dm = upward.where((upward > downward) & (upward > 0), 0.0)
    minus_dm = downward.where((downward > upward) & (downward > 0), 0.0)
    atr = _atr(dataframe, period).replace(0, float("nan"))
    plus_di = 100 * plus_dm.ewm(alpha=1 / period, adjust=False, min_periods=period).mean() / atr
    minus_di = 100 * minus_dm.ewm(alpha=1 / period, adjust=False, min_periods=period).mean() / atr
    denominator = (plus_di + minus_di).replace(0, float("nan"))
    dx = 100 * (plus_di - minus_di).abs() / denominator
    return dx.ewm(alpha=1 / period, adjust=False, min_periods=period).mean()


def _trend_age(condition: pd.Series) -> pd.Series:
    groups = condition.ne(condition.shift()).cumsum()
    age = condition.groupby(groups).cumcount() + 1
    return age.where(condition, 0)


class Strategy1(IStrategy):
    """20x multi-timeframe EMA pullback strategy for backtest and dry-run only."""

    INTERFACE_VERSION = 3
    can_short = True
    timeframe = "1m"
    startup_candle_count = 240
    process_only_new_candles = True
    minimal_roi = {"0": 0.15}
    # At 20x, two 0.05% taker fees add about 2 ROI points.  -0.23 therefore
    # targets approximately -25% net margin ROI, matching the desktop guard.
    stoploss = -0.23
    use_exit_signal = False
    exit_profit_only = False
    ignore_roi_if_entry_signal = False
    position_adjustment_enable = False
    max_entry_position_adjustment = 0
    initial_stake = 50.0

    def _project_value(self, pair, key, default):
        projects = self.config.get("bq_projects", {}) if hasattr(self, "config") else {}
        return (projects.get(pair) or {}).get(key, default)
    order_types = {
        "entry": "market",
        "exit": "market",
        "emergency_exit": "market",
        "force_entry": "market",
        "force_exit": "market",
        "stoploss": "market",
        "stoploss_on_exchange": False,
    }

    def leverage(self, pair, current_time, current_rate, proposed_leverage, max_leverage, entry_tag, side, **kwargs):
        return min(float(self._project_value(pair, "leverage", 20.0)), max_leverage)

    def custom_stake_amount(self, pair, current_time, current_rate, proposed_stake, min_stake, max_stake, leverage, entry_tag, side, **kwargs):
        stake = min(float(self._project_value(pair, "initial_stake", self.initial_stake)), max_stake)
        if min_stake is not None and stake < min_stake:
            return min_stake if min_stake <= max_stake else 0.0
        return stake

    @informative("5m")
    def populate_indicators_5m(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        dataframe["ema7"] = _ema(dataframe["close"], 7)
        dataframe["ema25"] = _ema(dataframe["close"], 25)
        dataframe["ema100"] = _ema(dataframe["close"], 100)
        dataframe["adx"] = _adx(dataframe)
        dataframe["atr"] = _atr(dataframe)
        dataframe["ema25_slope"] = dataframe["ema25"].pct_change(3)
        dataframe["trend_long"] = (dataframe["ema7"] > dataframe["ema25"]) & (dataframe["ema25"] > dataframe["ema100"]) & (dataframe["ema25_slope"] > 0) & (dataframe["adx"] >= 20.0)
        dataframe["trend_short"] = (dataframe["ema7"] < dataframe["ema25"]) & (dataframe["ema25"] < dataframe["ema100"]) & (dataframe["ema25_slope"] < 0) & (dataframe["adx"] >= 20.0)
        return dataframe

    @informative("15m")
    def populate_indicators_15m(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        dataframe["ema20"] = _ema(dataframe["close"], 20)
        dataframe["ema60"] = _ema(dataframe["close"], 60)
        dataframe["adx"] = _adx(dataframe)
        dataframe["atr"] = _atr(dataframe)
        dataframe["ema20_slope"] = dataframe["ema20"].pct_change(3)
        dataframe["trend_long"] = (dataframe["close"] > dataframe["ema60"]) & (dataframe["ema20"] > dataframe["ema60"]) & (dataframe["ema20_slope"] > 0) & (dataframe["adx"] >= 18.0)
        dataframe["trend_short"] = (dataframe["close"] < dataframe["ema60"]) & (dataframe["ema20"] < dataframe["ema60"]) & (dataframe["ema20_slope"] < 0) & (dataframe["adx"] >= 18.0)
        dataframe["trend_long_age"] = _trend_age(dataframe["trend_long"])
        dataframe["trend_short_age"] = _trend_age(dataframe["trend_short"])
        return dataframe

    def populate_indicators(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        dataframe["ema7"] = _ema(dataframe["close"], 7)
        dataframe["ema25"] = _ema(dataframe["close"], 25)
        dataframe["atr"] = _atr(dataframe)
        dataframe["volume_sma20"] = dataframe["volume"].rolling(20).mean()
        long_reclaim = (dataframe["close"].shift(1) <= dataframe["ema7"].shift(1)) & (dataframe["close"] > dataframe["ema7"]) & (dataframe["close"] > dataframe["open"])
        short_reclaim = (dataframe["close"].shift(1) >= dataframe["ema7"].shift(1)) & (dataframe["close"] < dataframe["ema7"]) & (dataframe["close"] < dataframe["open"])
        long_strength = dataframe["trend_long_5m"] & dataframe["trend_long_15m"] & (dataframe["adx_5m"] >= 30.0) & (dataframe["adx_15m"] >= 25.0) & ((dataframe["ema7_5m"] - dataframe["ema25_5m"]) / dataframe["close_5m"] >= 0.0008) & ((dataframe["ema20_15m"] - dataframe["ema60_15m"]) / dataframe["close_15m"] >= 0.0015)
        short_strength = dataframe["trend_short_5m"] & dataframe["trend_short_15m"] & (dataframe["adx_5m"] >= 25.0) & (dataframe["adx_15m"] >= 20.0) & ((dataframe["ema25_5m"] - dataframe["ema7_5m"]) / dataframe["close_5m"] >= 0.0008) & ((dataframe["ema60_15m"] - dataframe["ema20_15m"]) / dataframe["close_15m"] >= 0.0015)
        near_5m = (dataframe["close_5m"] - dataframe["ema7_5m"]).abs() <= dataframe["atr_5m"] * 0.80
        near_15m = (dataframe["close_15m"] - dataframe["ema20_15m"]).abs() <= dataframe["atr_15m"] * 1.50
        liquid = (dataframe["volume"] > 0) & (dataframe["volume"] >= dataframe["volume_sma20"])
        dataframe["entry_long_setup"] = long_strength & long_reclaim & ((dataframe["close"] - dataframe["ema25"]) <= dataframe["atr"] * 0.80) & near_5m & near_15m & dataframe["trend_long_age_15m"].between(1, 16) & liquid
        dataframe["entry_short_setup"] = short_strength & short_reclaim & ((dataframe["ema25"] - dataframe["close"]) <= dataframe["atr"] * 0.80) & near_5m & near_15m & dataframe["trend_short_age_15m"].between(1, 16) & liquid
        return dataframe

    def populate_entry_trend(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        long_event = dataframe["entry_long_setup"] & ~dataframe["entry_long_setup"].shift(1).fillna(False)
        short_event = dataframe["entry_short_setup"] & ~dataframe["entry_short_setup"].shift(1).fillna(False)
        dataframe.loc[long_event, ["enter_long", "enter_tag"]] = (1, "mtf_pullback_long")
        dataframe.loc[short_event, ["enter_short", "enter_tag"]] = (1, "mtf_pullback_short")
        return dataframe

    def populate_exit_trend(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        dataframe["exit_long"] = 0
        dataframe["exit_short"] = 0
        return dataframe
